pve：
	#导入虚拟机磁盘(img)
	qm disk import <vmid> <source> <storage> --format qcow2
	#创建swap文件并挂载
	cat /proc/sys/vm/swappiness && echo 'vm.swappiness = 20' > /etc/sysctl.conf && sysctl -p
	echo -e '0 0 * * * /bin/bash -c "swapoff -a && swapon -a"' | tee -a /var/spool/cron/crontabs/root
	fallocate -l 4G /mnt/swapfile && mkswap /mnt/swapfile && swapon /mnt/swapfile && echo '/mnt/swapfile none swap sw 0 0' >> /etc/fstab
	#修改网络桥配置自动获取
	echo -e "\nnet.ipv6.conf.all.forwarding = 1\nnet.ipv6.conf.vmbr0.accept_ra = 2" >> /etc/sysctl.conf
	echo -e "\niface vmbr0 inet6 auto\n\taccept_ra 2" >> /etc/network/interfaces && ifdown vmbr0 && ifup vmbr0
